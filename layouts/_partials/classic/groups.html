{{- /* Meditation groups (data/groups.yaml) in production order. Per group: h1 heading, h4 leader, description, roster table, image (optionally linked), more. Dividers as on production: one before each group except a group on the same day as the previous one (Monday Yoga follows Monday meditation without a divider), and one before the last group's `more` (the page's closing paragraphs). Classes per module: group- (per-group production styling in 30-pages.css), group--same-day (smaller h1, as the Yoga section). Context: the page. Returns HTML. */ -}} {{- $page := . -}} {{- $out := "" -}} {{- $prevDay := "" -}} {{- $groups := site.Data.groups.groups | default slice -}} {{- $last := sub (len $groups) 1 -}} {{- range $i, $g := $groups -}} {{- $day := $g.day | default "" -}} {{- $same := and (gt $i 0) (ne $day "") (eq $day $prevDay) -}} {{- $cls := printf "group-%s%s" (anchorize $g.name) (cond $same " group--same-day" "") -}} {{- $h := printf "

%s

\n" (anchorize $g.name) ($g.heading | default $g.name | htmlEscape) -}} {{- with $g.leader }}{{ $h = printf "%s

%s

\n" $h ($page.RenderString .) }}{{ end -}} {{- with $g.description }}{{ $h = printf "%s%s\n" $h ($page.RenderString (dict "display" "block") .) }}{{ end -}} {{- with $g.roster -}} {{- $rows := "" -}} {{- range . }}{{ $rows = printf "%s%s%s\n" $rows ((time.AsTime .date).Format "January 2") (.teacher | htmlEscape) }}{{ end -}} {{- $h = printf "%s\n\n%s\n
\n" $h $rows -}} {{- end -}} {{- $mod := printf "
%s
" $cls $h -}} {{- with $g.image -}} {{- $img := printf "" (partial "url.html" .) (partial "img-size.html" .) ($g.image_alt | default "" | htmlEscape) -}} {{- with $g.image_link }}{{ $img = printf "%s" (partial "url.html" .) $img }}{{ end -}} {{- $mod = printf "%s\n
%s
" $mod (printf "group-%s" (anchorize $g.name)) $img -}} {{- end -}} {{- with $g.more -}} {{- if eq $i $last }}{{ $mod = printf "%s\n%s" $mod (partial "classic/divider.html" .) }}{{ end -}} {{- $mod = printf "%s\n
%s
" $mod (printf "group-%s" (anchorize $g.name)) ($page.RenderString (dict "display" "block") .) -}} {{- end -}} {{- if or (eq $i 0) (ne $day $prevDay) (eq $day "") }}{{ $out = printf "%s\n%s" $out (partial "classic/divider.html" .) }}{{ end -}} {{- $out = printf "%s\n%s" $out $mod -}} {{- $prevDay = $day -}} {{- end -}} {{- return ($out | safeHTML) -}}
